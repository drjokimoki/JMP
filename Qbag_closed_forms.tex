\begin{center}
\textbf{\large Closed Forms and Functional-Form Approximations for $\Qbag$}
\end{center}

\noindent
The infinite-bagged excess risk is governed by the \emph{conditional} harmonic mean
\begin{equation}\label{eq:gdef}
  g_i(k)\;=\;k\,\E_S\!\left[\frac{1}{A_S}\,\Bigg|\,i\in S\right],
  \qquad
  \Rk \;=\; \frac{1}{m^2}\sum_{i=1}^m \tau_i\!\left(g_i(k)-\frac{1}{\bar\tau}\right)^{\!2}.
\end{equation}
Unlike the unconditional $\E_S[1/A_S]$ that yields $\Qsub$, the conditioning $i\in S$
blocks a general closed form. It does not, however, block the following.

%==========================================================================
\section*{1. Exact: homoskedastic idiosyncratic block (equicorrelation)}

Let the common-loading covariance specialise to a homoskedastic idiosyncratic block,
\begin{equation}\label{eq:equicorr}
  \Sig \;=\; \sigma^2\big[(1-\rho)\,I_m + \rho\,\iotab\iotab'\big]
  \;=\; D + q\,\iotab\iotab',
  \qquad D=\sigma^2(1-\rho)I_m,\quad q=\sigma^2\rho .
\end{equation}

\begin{proposition}[Exact bagged risk under equicorrelation]\label{prop:equi}
Under \eqref{eq:equicorr}, every size-$k$ subset is exchangeable, so $w_S=\iotab/k$ and
\begin{equation}
  Q(S)=\sigma^2\!\left[\rho+\frac{1-\rho}{k}\right]\quad\text{(independent of }S\text{),}
\end{equation}
hence
\begin{equation}\label{eq:equi-result}
  \Qsub=\Qbag=\sigma^2\!\left[\rho+\frac{1-\rho}{k}\right],
  \qquad
  \boxed{\;\Rk=\sigma^2(1-\rho)\,z_k\;}\quad\Longrightarrow\quad \alpha=1,
\end{equation}
with $z_k=\tfrac1k-\tfrac1m$ and $\Qopt=\sigma^2[\rho+(1-\rho)/m]$. No Monte Carlo is required.
\end{proposition}

\noindent\emph{Implication.} All analytical difficulty, and all of $\alpha>1$, arises
solely from heteroskedasticity in $\{\tau_i\}$ (i.e.\ a non-scalar block $D$).

%==========================================================================
\section*{2. Exact: finite-support precisions}

Suppose $\{\tau_i\}$ takes $G$ distinct values on groups of sizes $m_1,\dots,m_G$.
Then $A_S$ depends only on the group-count vector, which is multivariate hypergeometric,
so $g_i(k)$ is a finite sum. For the two-group case
($\tau_H$ on $m_H$ forecasters, $\tau_L$ on the remaining $m-m_H$):
\begin{equation}\label{eq:twogroup}
  g_H(k)\;=\;k\sum_{n=0}^{k-1}
  \frac{\dbinom{m_H-1}{n}\dbinom{m-m_H}{\,k-1-n\,}}{\dbinom{m-1}{k-1}}
  \cdot\frac{1}{\tau_H+n\,\tau_H+(k-1-n)\,\tau_L},
\end{equation}
and symmetrically for $g_L(k)$ (swap roles, use $m_L=m-m_H$). Substituting into
\eqref{eq:gdef} gives $\Rk$ exactly. The case $m=2$ is the trivial sub-anchor.
This is the appropriate tool in the dominant-forecaster regime, where the
Taylor approximation of \S3 loses accuracy.

%==========================================================================
\section*{3. Functional-form approximation: general heteroskedastic $D$}

The leave-one-out sum $A_{S\setminus\{i\}}=\sum_{j\in S,\,j\neq i}\tau_j$ is the sum of
$k-1$ precisions drawn without replacement from the $m-1$ values $\{\tau_j\}_{j\neq i}$,
with hypergeometric mean and variance
\begin{equation}\label{eq:loo-moments}
  \mu_i=(k-1)\,\bar\tau_{-i},
  \qquad
  V_i=(k-1)\,\frac{m-k}{(m-1)(m-2)}\sum_{j\neq i}(\tau_j-\bar\tau_{-i})^2,
  \qquad
  \bar\tau_{-i}=\frac{A_m-\tau_i}{m-1}.
\end{equation}

\begin{proposition}[Second-order leave-one-out expansion]\label{prop:mf2}
A second-order expansion of $1/(\tau_i+A_{S\setminus\{i\}})$ about $A_{S\setminus\{i\}}=\mu_i$ gives
\begin{equation}\label{eq:mf2}
  \boxed{\;
  g_i(k)\;\approx\;\frac{k}{\tau_i+\mu_i}
  \left[\,1+\frac{V_i}{(\tau_i+\mu_i)^2}\,\right],
  \qquad
  \Rk\;\approx\;\frac{1}{m^2}\sum_{i=1}^m \tau_i\!\left(g_i(k)-\frac{1}{\bar\tau}\right)^{\!2}.
  \;}
\end{equation}
\end{proposition}

\noindent\emph{Small-heterogeneity limit.} Writing $\tau_i=\bar\tau(1+\delta_i)$ with
$\sum_i\delta_i=0$ and $V_\delta=\tfrac1m\sum_i\delta_i^2$, equation \eqref{eq:mf2} collapses to
\begin{equation}\label{eq:smallhet}
  \Rk\;\approx\;\frac{V_\delta}{m\,\bar\tau}\,\frac{1}{k^2}
  \;=\;\frac{V_\delta}{m\,\bar\tau}\,z_k^2\,\bigl(1+o(1)\bigr),
  \qquad \alpha=2,
\end{equation}
recovering the H\'ajek-projection exponent with an explicit constant.
The first-order term alone ($V_i\equiv0$) is exact only at $k=1$ and understates the
decay; the variance term $V_i$ supplies the curvature that sets $\alpha$.

%==========================================================================
\section*{4. Continuous closed form: Gamma precisions}

Modelling $\tau_j\stackrel{iid}{\sim}\mathrm{Gamma}(a,b)$ (rate $b$, mean $\bar\tau=a/b$)
and Poissonising gives $A_{S\setminus\{i\}}\sim\mathrm{Gamma}(\nu,b)$ with $\nu=(k-1)a$, so
\begin{equation}\label{eq:gamma}
  g_i(k)\;=\;k\,\E\!\left[\frac{1}{\tau_i+A_{S\setminus\{i\}}}\right]
  \;=\;k\,b^{\nu}\,\tau_i^{\,\nu-1}\,U\!\big(\nu,\nu,\,b\tau_i\big),
  \qquad \nu=(k-1)a,
\end{equation}
where $U(\cdot,\cdot,\cdot)$ is the Tricomi confluent hypergeometric function. This is a
genuine functional form; it differs from the fixed-$\{\tau_i\}$ value because it
additionally integrates over $\tau_i$.

%==========================================================================
\section*{Numerical validation of Proposition~\ref{prop:mf2}}

Effective exponent $\alpha$ (log--log fit of $\Rk$ on $z_k$) and median level ratio,
exact all-subsets enumeration vs.\ \eqref{eq:mf2}:
\begin{center}
\begin{tabular}{lccc}
\hline
profile & $\alpha$ (exact) & $\alpha$ \eqref{eq:mf2} & median ratio \\
\hline
mild heterogeneity        & 1.99 & 1.99 & 1.00 \\
lognormal($0.5$)          & 2.04 & 2.04 & 0.99 \\
two-group                 & 1.84 & 1.86 & 0.91 \\
strong lognormal($1.0$)   & 1.59 & 1.74 & 0.79 \\
dominant ($10$ vs $1$)    & 1.25 & 1.49 & 0.86 \\
\hline
\end{tabular}
\end{center}
Accuracy is essentially exact for moderate heterogeneity and degrades only in the
heavy-tail/dominant regime, where the exact finite-support form \eqref{eq:twogroup}
should be used instead. The ceiling $\alpha=2$ holds for diagonal $D$; correlation
structure in $\Sig$ (factor, Toeplitz) can push the fitted exponent above $2$.

%==========================================================================
\section*{5. Approximations from existing theory}

The building block $m\,\bar v_{k,i}=\tau_i\,g_i(k)$ is a simple-random-sample (without
replacement) average of a reciprocal. Four established literatures give handles on it.

\subsection*{5.1 Negative-moment (Laplace) identity: exact integral form}

Since $A_S>0$, the elementary identity $\E[1/X]=\int_0^\infty\E[e^{-tX}]\,dt$ converts the
reciprocal into a moment generating function. Over uniform size-$k$ subsets this is an
\emph{elementary symmetric polynomial} $e_k$ in $x_j=e^{-t\tau_j}$:
\begin{align}
  \E_S[1/A_S]
  &= \binom{m}{k}^{-1}\!\int_0^\infty e_k\!\big(e^{-t\tau_1},\dots,e^{-t\tau_m}\big)\,dt,
  \label{eq:lap-uncond}\\[2pt]
  g_i(k)
  &= k\,\binom{m-1}{k-1}^{-1}\!\int_0^\infty e^{-t\tau_i}\,
     e_{k-1}\!\big((e^{-t\tau_j})_{j\neq i}\big)\,dt .
  \label{eq:lap-cond}
\end{align}
Both are exact. In the i.i.d.\ (Poissonised) limit, with negative-MGF
$\phi(t)=\E[e^{-t\tau}]$, the conditional form factorises:
\begin{equation}\label{eq:lap-iid}
  g_i(k)\;\approx\;k\int_0^\infty e^{-t\tau_i}\,\phi(t)^{\,k-1}\,dt .
\end{equation}
For $\tau\sim\mathrm{Gamma}(a,b)$, $\phi(t)=(1+t/b)^{-a}$ and \eqref{eq:lap-iid} integrates to
the Tricomi form $g_i(k)=k\,b\,U\!\big(1,\,2-\nu,\,b\tau_i\big)=k\,b^{\nu}\tau_i^{\nu-1}U(\nu,\nu,b\tau_i)$,
$\nu=(k-1)a$, recovering eq.~\eqref{eq:gamma} as a special case.

\subsection*{5.2 Finite-population ratio-estimator bias (Cochran)}

$g_i(k)$ is the conditional mean of a reciprocal under simple random sampling without
replacement; the standard second-order ratio-estimator bias expansion
(Cochran, \emph{Sampling Techniques}, 1977) reproduces Proposition~\ref{prop:mf2}
exactly, with moments \eqref{eq:loo-moments} and a remainder
\begin{equation}\label{eq:cochran-rem}
  \Big|\,g_i(k)-\tfrac{k}{\tau_i+\mu_i}\big[1+V_i/(\tau_i+\mu_i)^2\big]\,\Big|
  \;=\;O\!\left(\frac{k\,\E|A_{S\setminus\{i\}}-\mu_i|^3}{(\tau_i+\mu_i)^4}\right).
\end{equation}

\subsection*{5.3 U-statistic / Hoeffding--H\'ajek decomposition and bagging theory}

$m\,\bar v_{k,i}$ is a U-statistic of degree $k$ with kernel $\tau_i/A_S$
(Hoeffding 1948). Its first-order (H\'ajek) projection yields the leading term
\begin{equation}\label{eq:hajek}
  \Rk \;=\; \underbrace{\frac{V_\delta}{m\,\bar\tau}\,z_k^2}_{\text{H\'ajek projection,}\ \alpha=2}
  \;+\;\sum_{c\geq 2}\zeta_c(k),
  \qquad V_\delta=\tfrac1m\textstyle\sum_i\delta_i^2,\ \ \tau_i=\bar\tau(1+\delta_i),
\end{equation}
where $\zeta_c(k)$ are the higher Hoeffding ANOVA variance components; these bend the
effective exponent away from $2$ under strong heterogeneity. This is the formal basis of
``$\alpha=2$ is the diagonal ceiling'' (Bühlmann \& Yu 2002, \emph{Ann.\ Statist.};
Buja \& Stuetzle 2006; Friedman \& Hall 2007).

\subsection*{5.4 Concentration under sampling without replacement}

Bernstein-type inequalities for SRSWOR (Serfling 1974; Bardenet \& Maillard 2015) bound
$\Pr(|A_S-k\bar\tau|>s)\le 2\exp\{-s^2/(2k\sigma_\tau^2+\tfrac23 s\,\tau_{\max})\}$,
which controls the Taylor remainder and upgrades the expansions of \S3 and \S5.2 from
approximations to bounds with explicit error.

\subsection*{5.5 Citable anchors}

The equicorrelation result \eqref{eq:equi-result} is the textbook diversification / global
minimum-variance portfolio formula (cf.\ Kan \& Zhou 2007 for the estimated-$\Sig$ analogue),
and the finite-$G$ gap $G^{-1}(\Qsub-\Qbag)$ is exactly the variance of Shapley/coalition
sampling at coalition size $k$ (Covert \& Lee 2021).

\section*{6. The weight-deviation view: dropout, RandNLA, and shrinkage}

Adopt the weight-deviation definition
\begin{equation}\label{eq:wdev}
  \Rk=(\bar v_k-w^*)'\Sig\,(\bar v_k-w^*)=\sum_{i=1}^m\tau_i^{-1}\big(\bar v_{k,i}-w_i^*\big)^2,
  \qquad w_i^*=\frac{\tau_i}{m\bar\tau}.
\end{equation}
Then $\Rk$ is the squared bias of a \emph{randomly sparsified} estimator of the oracle
$w^*$. The bagged weight is a coordinate-wise shrinkage of $w^*$,
\begin{equation}\label{eq:shrink}
  \bar v_{k,i}=\bar\tau\,g_i(k)\,w_i^*,
  \qquad \bar v_{1}=\tfrac1m\iotab\ \text{(uniform)},\quad \bar v_{m}=w^*\ \text{(oracle)},
\end{equation}
so $\bar v_k$ interpolates uniform $\to$ oracle as $k:1\to m$, and \eqref{eq:wdev} is the
squared shrinkage bias. Three literatures approximate this object directly.

\subsection*{6.1 Dropout as adaptive regularization}

RSM is structured dropout on the combination weights (keep $k$ of $m$). In the
Bernoulli-keep relaxation (keep probability $p=k/m$ independently per coordinate; this
concentrates at fixed size $k$ as $m\to\infty$), the expected dropout solution is a
ridge-shrunk oracle (Wager, Wang \& Liang 2013; Baldi \& Sadowski 2013):
\begin{equation}\label{eq:dropout}
  \bar v_k\;\approx\;\frac{(\Sig+\Lambda)^{-1}\iotab}{\iotab'(\Sig+\Lambda)^{-1}\iotab},
  \qquad \Lambda=\frac{1-p}{p}\,\mathrm{diag}(\Sig).
\end{equation}
Under the budget constraint $\iotab'w=1$ the isotropic part of $\Lambda$ cancels, so the
bias is driven solely by the heterogeneity of $\mathrm{diag}(\Sig)$, reproducing
$\Rk=O\!\big((1-p)^2\big)$, i.e.\ $\alpha=2$, and the rule ``homoskedastic $D\Rightarrow$ no
bias.'' This supplies the regularization interpretation but no predictive content beyond
\S3.

\subsection*{6.2 Randomized linear algebra / leverage-score sampling}

$w_S$ is the constrained solution on a random size-$k$ column subset, and $\Rk$ is the
expected squared reconstruction error. Randomized-NLA theory (Drineas \& Mahoney 2006;
Boutsidis, Mahoney \& Drineas 2009) shows uniform sampling is variance-suboptimal: the
minimizing importance distribution draws coordinate $i$ with probability proportional to
its leverage, here $p_i\propto\tau_i$. The penalty paid by \emph{uniform} (flat-prior) RSM
is exactly the precision dispersion
\begin{equation}\label{eq:leverage}
  V_\delta=\frac{\mathrm{Var}(\tau)}{\bar\tau^2}=\frac1m\sum_i\delta_i^2,
  \qquad \Rk\;\approx\;\frac{V_\delta}{m\,\bar\tau}\,z_k^2,
\end{equation}
matching \eqref{eq:smallhet}/\eqref{eq:hajek}. This motivates an
\emph{importance-sampled RSM} ($p_i\propto\tau_i$) with strictly smaller $\Rk$, and shows
the flat subset prior of \S(BMA comparison) is not the variance-optimal sampler.

\subsection*{6.3 Shrinkage estimators and a spectral bound}

By \eqref{eq:shrink}, $\bar v_k$ is a coordinate-heterogeneous shrinkage of $w^*$ toward
the uniform vector $\tfrac1m\iotab$; linear-shrinkage MSE results (Ledoit \& Wolf;
James--Stein) provide an envelope on $\Rk$ (loose, as RSM's shrinkage is non-scalar).
Finally, since $\bar v_k-w^*$ lives in the span of $\Sig$, $\Rk$ is bounded by the tail
eigenvalue mass of $\Sig$,
\begin{equation}\label{eq:spectral}
  \Rk\;\le\;\lambda_{\max}(\Sig)\,\|\bar v_k-w^*\|_2^2,
\end{equation}
linking the geometric decay observed for the Toeplitz DGP to the $\Sig$ spectrum.

\subsection*{6.4 What leverage sampling pins down for \emph{uniform} RSM}

The leverage framework does more than flag suboptimality; it \emph{derives} $\Rk$ for our
case via two identifications.

\paragraph{(i) Oracle weight $=$ leverage score.}
In the common-loading model the constrained-GLS influence of forecaster $i$ is
\begin{equation}\label{eq:leverage-id}
  \ell_i \;=\; \frac{\tau_i}{A_m}\;=\;w_i^*,
\end{equation}
so the statistical leverage scores are exactly the oracle weights, and the
variance-optimal RandNLA proposal $p_i\propto\ell_i$ is ``sample in proportion to $w^*$.''

\paragraph{(ii) Uniform RSM is self-normalized importance sampling.}
A single size-$k$ draw estimates $w_i^*$ by the self-normalized ratio
$\tau_i/A_S$ (unnormalized weight $\tau_i$, normalized over the drawn set $S$);
infinite bagging Rao--Blackwellizes it,
\begin{equation}\label{eq:snis}
  \bar v_{k,i}=\E_S\!\left[\frac{\tau_i\,\mathbf 1_{i\in S}}{A_S}\right],
\end{equation}
which is the Rao--Blackwellized self-normalized (SNIS) estimator of the leverage
vector under a \emph{uniform} proposal. The SNIS bias equals the ratio-estimator bias of
\S5.2; thus leverage theory \emph{derives} $\Rk$ but does not improve on \S3.

\paragraph{(iii) Sandwich bound and the role of coherence.}
Combining the bagging-$\le$-subbagging inequality with the Taylor expansion of
$\E_S[1/A_S]$ (\S4.1 of the paper) and the diagonal H\'ajek ceiling \eqref{eq:hajek}:
\begin{equation}\label{eq:sandwich}
  \underbrace{\frac{V_\delta}{m\bar\tau}\,z_k^2\,(1+o(1))}_{\text{diagonal H\'ajek},\ \alpha=2}
  \;\lesssim\;\Rk\;\le\;Q_k^{\mathrm{sub}}-\Qopt\;\le\;\frac{z_k}{\bar\tau}\,(1+o(1))
  \;\;\Longrightarrow\;\; 1\le\alpha\le 2 .
\end{equation}
Where $\alpha$ falls in $[1,2]$ is governed by the \emph{precision coherence}
\begin{equation}\label{eq:coherence}
  \kappa\;=\;\frac{\tau_{\max}}{\bar\tau}\;=\;m\,\max_i \ell_i :
\end{equation}
low coherence $\Rightarrow$ uniform $\approx$ leverage sampling $\Rightarrow \alpha\to2$;
high coherence (a dominant forecaster) $\Rightarrow$ uniform is far from the
variance-optimal proposal $\Rightarrow \alpha\to1$. This matches the empirical
dominant-forecaster regime. The left bound in \eqref{eq:sandwich} is the leading-order
value under mild heterogeneity, not a uniform lower bound; the right inequalities are exact.

\paragraph{Numerical validation of \eqref{eq:coherence}.}
Effective exponent $\alpha$ (log--log fit of the exact all-subsets $\Rk$ on $z_k$,
$m=9$) against the precision coherence $\kappa=\tau_{\max}/\bar\tau$:
\begin{center}
\begin{tabular}{lcc}
\hline
precision profile & coherence $\kappa$ & $\alpha$ \\
\hline
near-homogeneous          & 1.09 & 2.00 \\
mild heterogeneity        & 1.37 & 2.01 \\
lognormal($0.5$)          & 1.48 & 2.09 \\
lognormal($1.0$)          & 2.37 & 1.75 \\
two-group                 & 2.65 & 1.80 \\
dominant                  & 5.40 & 1.24 \\
very dominant             & 7.50 & 1.08 \\
\hline
\end{tabular}
\end{center}
The map $\kappa\downarrow\,\Rightarrow\,\alpha\uparrow$ is monotone across the range:
$\kappa\to1$ gives $\alpha\to2$ (uniform sampling near-optimal), large $\kappa$ gives
$\alpha\to1$ (uniform far from the leverage-optimal proposal). Reproduced by
\texttt{coherence\_alpha\_check.py}.
